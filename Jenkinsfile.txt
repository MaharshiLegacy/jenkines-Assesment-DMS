pipeline {
    agent any
    tools{
      jdk 'java21'
    }

  
    
    stages{
         stage('Check Java Version') {
    steps {
        sh 'chmod +x mvnw'
        sh 'echo "JAVA_HOME=$JAVA_HOME"'
        sh 'java -version'
        sh './mvnw -version'
    }
}
      stage('Checkout') {
              steps {
                  checkout scm
              }
          }

        stage('Check Java Compiler') {
    steps {
        sh 'echo "JAVA_HOME=$JAVA_HOME"'
        sh 'which java'
        sh 'java -version'
        sh 'which javac'
        sh 'javac -version'
        sh 'javac --help | grep -A 2 -- "--release"'
    }
}
        
    stage('Build and Test') {
    steps {
        sh 'chmod +x mvnw'
        sh './mvnw clean verify'
    }
}

        
stage('Build Docker Image') {
    steps {
        sh '''
            set -eu

            PROJECT_ID="project-f45bd8e2-3bd4-49b8-a29"
            REGION="asia-south1"
            REPOSITORY="jenkins-dms-repository"
            IMAGE_NAME="dms-image"

            IMAGE_URI="${REGION}-docker.pkg.dev/${PROJECT_ID}/${REPOSITORY}/${IMAGE_NAME}"

            echo "Building Docker image: ${IMAGE_URI}:${BUILD_NUMBER}"

            docker build \
                -t "${IMAGE_URI}:${BUILD_NUMBER}" \
                -t "${IMAGE_URI}:latest" .
        '''
    }
}

stage('Push Image to Artifact Registry') {
    steps {
        sh '''
            set -eu

            PROJECT_ID="project-f45bd8e2-3bd4-49b8-a29"
            REGION="asia-south1"
            REPOSITORY="jenkins-dms-repository"
            IMAGE_NAME="dms-image"

            IMAGE_URI="${REGION}-docker.pkg.dev/${PROJECT_ID}/${REPOSITORY}/${IMAGE_NAME}"

            gcloud auth configure-docker \
                "${REGION}-docker.pkg.dev" --quiet

            docker push "${IMAGE_URI}:${BUILD_NUMBER}"
            docker push "${IMAGE_URI}:latest"

            echo "Image pushed successfully."
            echo "${IMAGE_URI}:${BUILD_NUMBER}"
        '''
    }
}

        
stage('Deploy to GKE') {
    steps {
        sh '''
            set -eu

            PROJECT_ID="project-f45bd8e2-3bd4-49b8-a29"
            REGION="asia-south1"
            CLUSTER="jenkins-gke-cluster"
            REPOSITORY="jenkins-dms-repository"
            IMAGE_NAME="dms-image"
            IMAGE_URI="${REGION}-docker.pkg.dev/${PROJECT_ID}/${REPOSITORY}/${IMAGE_NAME}:${BUILD_NUMBER}"

            gcloud config set project "${PROJECT_ID}"

            gcloud container clusters get-credentials "${CLUSTER}" \
              --region="${REGION}" \
              --project="${PROJECT_ID}"

            kubectl create namespace demo \
              --dry-run=client -o yaml | kubectl apply -f -

            # Save the existing deployment revision for rollback.
            if kubectl get deployment gke-demo -n demo >/dev/null 2>&1; then
                echo "Existing deployment found."
            else
                echo "First deployment: Kubernetes will create the Deployment."
            fi

            # Apply the manifest and use this build's unique image tag.
            sed "s|image: .*|image: ${IMAGE_URI}|" k8s/deployment.yaml \
              | kubectl apply -f -

            kubectl rollout status deployment/gke-demo \
              -n demo --timeout=300s
        '''
    }
}

stage('Verify Deployment') {
    steps {
        sh '''
            set -eu
            kubectl get deployment gke-demo -n demo
            kubectl get pods -n demo -o wide
            kubectl get deployment gke-demo -n demo \
              -o jsonpath='{.spec.template.spec.containers[0].image}'
            echo
        '''
    }
}


        
    
    }
}
